cmake_minimum_required(VERSION 3.28)
project(Shark VERSION 0.1.0 LANGUAGES CXX)

# Shark uses headers, not C++ module import units. Selecting C++20 must not
# require clang-scan-deps; Ninja still tracks ordinary compiler/header inputs.
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build: cmake --preset <preset>")
endif()

option(SHARK_BUILD_CLIENT "Build EngineClient, Engine composition and Runtime" ON)
option(SHARK_BUILD_SERVER "Build the headless Server" ON)
option(SHARK_BUILD_EDITOR "Build the editor foundation" ON)
option(SHARK_BUILD_TOOLS "Build CPU authoring services without requiring an editor or client" ${SHARK_BUILD_EDITOR})
option(SHARK_BUILD_TESTS "Build and register Shark's CPU tests" ON)
option(SHARK_BUILD_GRAPHICS_PROBE "Build the opt-in assignment 02 graphics probe" OFF)
set(shark_native_window_default OFF)
if(WIN32 AND SHARK_BUILD_CLIENT)
    set(shark_native_window_default ON)
endif()
option(SHARK_NATIVE_WINDOW "Enable the Windows GLFW backend in EngineClient" ${shark_native_window_default})
set(shark_native_graphics_default OFF)
if(WIN32 AND SHARK_BUILD_CLIENT AND SHARK_NATIVE_WINDOW)
    set(shark_native_graphics_default ON)
endif()
option(SHARK_NATIVE_GRAPHICS "Enable the Windows Vulkan/NVRHI backend in EngineClient" ${shark_native_graphics_default})
option(SHARK_INJECT_TEST_FAILURE "Deliberately fail the CTest failure sentinel" OFF)
set(SHARK_SANITIZER "none" CACHE STRING "Sanitizer: none, address, address-undefined")
set_property(CACHE SHARK_SANITIZER PROPERTY STRINGS none address address-undefined)
if(SHARK_INJECT_TEST_FAILURE AND NOT SHARK_BUILD_TESTS)
    message(FATAL_ERROR "SHARK_INJECT_TEST_FAILURE requires SHARK_BUILD_TESTS=ON")
endif()
if(SHARK_BUILD_EDITOR AND NOT SHARK_BUILD_CLIENT)
    message(FATAL_ERROR "SHARK_BUILD_EDITOR requires SHARK_BUILD_CLIENT=ON")
endif()
if(SHARK_BUILD_EDITOR AND NOT SHARK_BUILD_TOOLS)
    message(FATAL_ERROR "SHARK_BUILD_EDITOR requires SHARK_BUILD_TOOLS=ON")
endif()
if(SHARK_BUILD_GRAPHICS_PROBE AND NOT SHARK_BUILD_CLIENT)
    message(FATAL_ERROR "SHARK_BUILD_GRAPHICS_PROBE requires SHARK_BUILD_CLIENT=ON")
endif()
if(SHARK_NATIVE_GRAPHICS AND NOT SHARK_BUILD_CLIENT)
    message(FATAL_ERROR "SHARK_NATIVE_GRAPHICS requires SHARK_BUILD_CLIENT=ON")
endif()
if(SHARK_NATIVE_GRAPHICS AND NOT WIN32)
    message(FATAL_ERROR "SHARK_NATIVE_GRAPHICS is supported only on Windows")
endif()
if(SHARK_NATIVE_GRAPHICS AND NOT SHARK_NATIVE_WINDOW)
    message(FATAL_ERROR "SHARK_NATIVE_GRAPHICS requires SHARK_NATIVE_WINDOW=ON")
endif()
if(SHARK_NATIVE_WINDOW AND NOT SHARK_BUILD_CLIENT)
    message(FATAL_ERROR "SHARK_NATIVE_WINDOW requires SHARK_BUILD_CLIENT=ON")
endif()
if(SHARK_NATIVE_WINDOW AND NOT WIN32)
    message(FATAL_ERROR "SHARK_NATIVE_WINDOW is supported only on Windows")
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(SharkOptions)
include(SharkBoundaries)
shark_validate_sanitizer()
include(SceneDependencies)
shark_validate_scene_inputs()
include(MathDependencies)
shark_validate_math_inputs()
include(NetworkDependencies)
shark_validate_network_inputs()
if(SHARK_BUILD_TOOLS)
    include(SourceDocumentDependencies)
    shark_validate_source_document_inputs()
endif()
if(SHARK_NATIVE_GRAPHICS OR SHARK_BUILD_GRAPHICS_PROBE)
    include(GraphicsDependencies)
    shark_validate_graphics_inputs()
endif()
if(SHARK_NATIVE_WINDOW)
    include(WindowDependencies)
    shark_validate_window_inputs()
endif()
if(SHARK_BUILD_SERVER)
    include(StorageDependencies)
    shark_validate_storage_inputs()
endif()
# Private TLS/crypto compiles C in every Core configuration. Client C libraries
# remain separately gated by the native window/graphics options.
enable_language(C)
if(SHARK_NATIVE_WINDOW)
    shark_setup_window_dependencies()
endif()
if(SHARK_NATIVE_GRAPHICS OR SHARK_BUILD_GRAPHICS_PROBE)
    shark_setup_graphics_dependencies()
endif()

shark_setup_scene_dependencies()
shark_setup_math_dependencies()
shark_setup_network_dependencies()
add_subdirectory(engine)
if(SHARK_BUILD_CLIENT)
    add_subdirectory(runtime)
endif()
if(SHARK_BUILD_SERVER)
    shark_setup_storage_dependencies()
    add_subdirectory(server)
    install(TARGETS Server RUNTIME DESTINATION bin COMPONENT Server)
endif()
if(SHARK_BUILD_TOOLS)
    add_subdirectory(engine_tools)
endif()
if(SHARK_BUILD_EDITOR)
    add_subdirectory(editor)
endif()
if(SHARK_BUILD_TESTS)
    enable_testing()
    include(Dependencies)
    add_subdirectory(tests)
endif()
if(SHARK_BUILD_GRAPHICS_PROBE)
    add_subdirectory(probes/graphics)
endif()

shark_generate_boundary_check()
