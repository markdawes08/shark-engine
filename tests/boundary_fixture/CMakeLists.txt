cmake_minimum_required(VERSION 3.28)
project(SharkBoundaryFixture LANGUAGES NONE)
include("${SHARK_SOURCE}/cmake/SharkBoundaries.cmake")
foreach(shark_target IN ITEMS EngineCore Server Bridge)
    add_library(${shark_target} INTERFACE)
    shark_classify_target(${shark_target} core)
endforeach()
foreach(shark_target IN ITEMS Engine EngineClient Runtime)
    add_library(${shark_target} INTERFACE)
    shark_classify_target(${shark_target} client)
endforeach()
add_library(EngineTools INTERFACE)
shark_classify_target(EngineTools authoring)
target_link_libraries(Server INTERFACE EngineCore)
target_link_libraries(EngineCore INTERFACE Bridge)
target_link_libraries(EngineClient INTERFACE EngineCore)
target_link_libraries(Engine INTERFACE EngineClient)
target_link_libraries(Runtime INTERFACE Engine)

if(SHARK_BOUNDARY_CASE STREQUAL "client-link")
    target_link_libraries(Bridge INTERFACE EngineClient)
elseif(SHARK_BOUNDARY_CASE STREQUAL "client-genex")
    target_link_libraries(Bridge INTERFACE "$<IF:$<BOOL:1>,EngineClient,>")
elseif(SHARK_BOUNDARY_CASE STREQUAL "authoring-genex")
    target_link_libraries(Bridge INTERFACE "$<$<BOOL:1>:EngineTools>")
elseif(SHARK_BOUNDARY_CASE STREQUAL "runtime-authoring")
    target_link_libraries(Runtime INTERFACE "$<LINK_ONLY:EngineTools>")
elseif(SHARK_BOUNDARY_CASE STREQUAL "raw-link")
    target_link_libraries(Bridge INTERFACE "raw_unclassified_library")
elseif(SHARK_BOUNDARY_CASE STREQUAL "link-option")
    target_link_options(Bridge INTERFACE "$<$<BOOL:1>:/tmp/libclient.a>")
elseif(SHARK_BOUNDARY_CASE STREQUAL "security-flags")
    target_link_options(Bridge INTERFACE "LINKER:/guard:cf" "LINKER:/guard:ehcont")
elseif(SHARK_BOUNDARY_CASE STREQUAL "security-flag-injection")
    target_link_options(Bridge INTERFACE "LINKER:/guard:cf,/tmp/libclient.a")
elseif(SHARK_BOUNDARY_CASE STREQUAL "link-directory")
    target_link_directories(Bridge INTERFACE "${CMAKE_BINARY_DIR}/hidden")
elseif(SHARK_BOUNDARY_CASE STREQUAL "prebuilt-object")
    file(WRITE "${CMAKE_BINARY_DIR}/engine/core/hidden.o" "prebuilt object")
    target_sources(Bridge INTERFACE "${CMAKE_BINARY_DIR}/engine/core/hidden.o")
elseif(SHARK_BOUNDARY_CASE MATCHES "^utility-(client|authoring|unclassified)$")
    add_custom_target(GenerateCoreData)
    if(NOT SHARK_BOUNDARY_CASE STREQUAL "utility-unclassified")
        shark_classify_target(GenerateCoreData core)
        if(SHARK_BOUNDARY_CASE STREQUAL "utility-client")
            add_dependencies(GenerateCoreData EngineClient)
        else()
            add_dependencies(GenerateCoreData EngineTools)
        endif()
    endif()
    add_dependencies(Bridge GenerateCoreData)
elseif(SHARK_BOUNDARY_CASE MATCHES "^(include|source)-(client|authoring|json)$")
    if(SHARK_BOUNDARY_CASE MATCHES "client$")
        set(shark_leak "${CMAKE_BINARY_DIR}/engine/client/leak.hpp")
    elseif(SHARK_BOUNDARY_CASE MATCHES "json$")
        set(shark_leak "${CMAKE_BINARY_DIR}/_deps/json-src/include/nlohmann/json.hpp")
    else()
        set(shark_leak "${CMAKE_BINARY_DIR}/engine_tools/leak.hpp")
    endif()
    file(WRITE "${shark_leak}" "// A forbidden generated input.\n")
    if(SHARK_BOUNDARY_CASE MATCHES "^include")
        get_filename_component(shark_leak_dir "${shark_leak}" DIRECTORY)
        target_include_directories(Bridge INTERFACE "$<$<BOOL:1>:${shark_leak_dir}>")
    else()
        target_sources(Bridge INTERFACE "$<$<BOOL:1>:${shark_leak}>")
    endif()
elseif(SHARK_BOUNDARY_CASE STREQUAL "server-only")
    # A separate minimal fixture below is used so Runtime really does not exist.
    message(FATAL_ERROR "Use server_only fixture")
elseif(NOT SHARK_BOUNDARY_CASE STREQUAL "allowed")
    message(FATAL_ERROR "Unknown fixture case '${SHARK_BOUNDARY_CASE}'")
endif()
shark_generate_boundary_check()
